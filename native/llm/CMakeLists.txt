cmake_minimum_required(VERSION 3.22)
project(aurora_llm C CXX)

set(LLAMA_ROOT ${CMAKE_CURRENT_LIST_DIR}/../llama.cpp)
set(LLAMA_VERSION_MAJOR 0)
set(LLAMA_VERSION_MINOR 8)
set(LLAMA_VERSION_PATCH 0)
set(LLAMA_VERSION_BASE 0.8.0)
set(LLAMA_VERSION 0.8.0)
set(LLAMA_BUILD_COMMIT v0.8.0)
set(LLAMA_BUILD_NUMBER 0)
set(GGML_BUILD_NUMBER 0)
set(GGML_BUILD_COMMIT v0.8.0)
set(BUILD_SHARED_LIBS OFF)
set(GGML_BUILD_TESTS OFF)
set(GGML_BUILD_EXAMPLES OFF)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
    # dotprod is the device gate for the model, so every supported phone runs these kernels
    set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "" FORCE)
endif()

function(llama_add_compile_flags)
endfunction()

if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_compile_options(/utf-8 /bigobj)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif()

add_subdirectory(${LLAMA_ROOT}/ggml ${CMAKE_CURRENT_BINARY_DIR}/ggml EXCLUDE_FROM_ALL)
add_subdirectory(${LLAMA_ROOT}/src ${CMAKE_CURRENT_BINARY_DIR}/llama EXCLUDE_FROM_ALL)

add_library(aurora_llm SHARED aurora_llm.cpp aurora_whisper.cpp)
set_target_properties(aurora_llm PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_link_libraries(aurora_llm PRIVATE llama)
if(ANDROID)
    target_link_options(aurora_llm PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384)
else()
    find_package(JNI REQUIRED)
    target_link_libraries(aurora_llm PRIVATE JNI::JNI)
endif()

add_library(whisper STATIC ${CMAKE_CURRENT_LIST_DIR}/../whisper.cpp/src/whisper.cpp ${CMAKE_CURRENT_LIST_DIR}/../whisper.cpp/src/aneforge/whisper-aneforge.cpp)
set_target_properties(whisper PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_include_directories(whisper PUBLIC ${CMAKE_CURRENT_LIST_DIR}/../whisper.cpp/include)
target_compile_definitions(whisper PRIVATE WHISPER_VERSION="1.9.5")
target_link_libraries(whisper PUBLIC ggml)
target_link_libraries(aurora_llm PRIVATE whisper)
